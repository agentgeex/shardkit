function DraggableDialog({
  open, onOpenChange, title, description, children, footer,
  resizable = true, draggable = true, fullscreenable = true,
  defaultSize = { w: 560, h: 380 }, testid = "dialog",
}: {
  open: boolean;
  onOpenChange: (v: boolean) => void;
  title: string;
  description?: string;
  children: React.ReactNode;
  footer?: React.ReactNode;
  resizable?: boolean;
  draggable?: boolean;
  fullscreenable?: boolean;
  defaultSize?: DialogSize;
  testid?: string;
}) {
  const [pos, setPos] = React.useState({ x: 0, y: 0 });
  const [size, setSize] = React.useState<DialogSize>(defaultSize);
  const [full, setFull] = React.useState(false);
  const dragRef = React.useRef<{ sx: number; sy: number; ox: number; oy: number } | null>(null);
  const resizeRef = React.useRef<{ sx: number; sy: number; ow: number; oh: number } | null>(null);
  const [moved, setMoved] = React.useState(false);

  // 每次打开重置位置/全屏
  React.useEffect(() => {
    if (open) {
      setPos({ x: 0, y: 0 });
      setFull(false);
      setMoved(false);
    }
  }, [open]);

  const onDragStart = (e: React.PointerEvent) => {
    if (!draggable || full) return;
    (e.target as HTMLElement).setPointerCapture(e.pointerId);
    dragRef.current = { sx: e.clientX, sy: e.clientY, ox: pos.x, oy: pos.y };
  };
  const onDragMove = (e: React.PointerEvent) => {
    const d = dragRef.current;
    if (!d) return;
    const nx = d.ox + (e.clientX - d.sx);
    const ny = d.oy + (e.clientY - d.sy);
    if (!moved && Math.abs(nx - d.ox) + Math.abs(ny - d.oy) > 2) setMoved(true);
    setPos({ x: nx, y: ny });
  };
  const onDragEnd = () => { dragRef.current = null; };

  const onResizeStart = (e: React.PointerEvent) => {
    if (!resizable || full) return;
    e.stopPropagation();
    (e.target as HTMLElement).setPointerCapture(e.pointerId);
    resizeRef.current = { sx: e.clientX, sy: e.clientY, ow: size.w, oh: size.h };
  };
  const onResizeMove = (e: React.PointerEvent) => {
    const r = resizeRef.current;
    if (!r) return;
    setSize({
      w: Math.max(360, r.ow + (e.clientX - r.sx)),
      h: Math.max(220, r.oh + (e.clientY - r.sy)),
    });
  };
  const onResizeEnd = () => { resizeRef.current = null; };

  const style: React.CSSProperties = full
    ? { width: "calc(100vw - 32px)", height: "calc(100vh - 32px)", maxWidth: "none" }
    : {
        width: size.w,
        height: size.h,
        maxWidth: "none",
        transform: `translate(calc(-50% + ${pos.x}px), calc(-50% + ${pos.y}px))`,
      };

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent
        data-testid={testid}
        onPointerMove={onDragMove}
        onPointerUp={onDragEnd}
        className="gap-0 overflow-hidden p-0 [&>.absolute.right-4.top-4]:hidden"
        style={style}
        onPointerDownOutside={(e) => e.preventDefault()}
        onInteractOutside={(e) => e.preventDefault()}
      >
        {/* 头部：标题 + 全屏/关闭 */}
        <DialogHeader
          onPointerDown={onDragStart}
          className={`flex h-12 flex-row items-center justify-between border-b px-4 py-0 pr-4 ${
            draggable && !full ? "cursor-move select-none" : ""
          }`}
        >
          <div className="flex items-center gap-2">
            {draggable && !full ? <GripHorizontal className="h-4 w-4 text-muted-foreground" /> : null}
            <DialogTitle className="text-base">{title}</DialogTitle>
          </div>
          <div className="flex items-center gap-1">
            {fullscreenable ? (
              <button
                type="button"
                aria-label={full ? "退出全屏" : "全屏"}
                data-testid={`${testid}-fullscreen`}
                onClick={() => setFull((f) => !f)}
                className="flex h-7 w-7 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
              >
                {full ? <Minimize2 className="h-4 w-4" /> : <Maximize2 className="h-4 w-4" />}
              </button>
            ) : null}
            <button
              type="button"
              aria-label="关闭"
              data-testid={`${testid}-close`}
              onClick={() => onOpenChange(false)}
              className="flex h-7 w-7 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
            >
              <span className="text-lg leading-none">×</span>
            </button>
          </div>
        </DialogHeader>

        {/* 内容区：可滚动 */}
        <DialogBody className="p-4" data-testid={`${testid}-body`}>
          {description ? (
            <DialogDescription className="mb-3">{description}</DialogDescription>
          ) : null}
          {children}
        </DialogBody>

        {/* 底部 */}
        {footer ? (
          <DialogFooter className="flex-row justify-end gap-2 space-x-0 border-t px-4 py-3">
            {footer}
          </DialogFooter>
        ) : null}

        {/* 缩放手柄 */}
        {resizable && !full ? (
          <div
            data-testid={`${testid}-resize`}
            onPointerDown={onResizeStart}
            onPointerMove={onResizeMove}
            onPointerUp={onResizeEnd}
            className="absolute bottom-0 right-0 h-4 w-4 cursor-nwse-resize"
            style={{
              background:
                "linear-gradient(135deg, transparent 0 50%, hsl(var(--muted-foreground) / 0.4) 50% 60%, transparent 60% 70%, hsl(var(--muted-foreground) / 0.4) 70% 80%, transparent 80%)",
            }}
          />
        ) : null}
      </DialogContent>
    </Dialog>
  );
}

import * as React from "react";
import { cn } from "@/lib/utils";

/**
 * ██ PageHeader —— 页头
 * ────────────────────────────────────────────────────────────────────────
 * 2026-10-28 采纳《页面排版-白框模式》规范 §4，新增 `layout` 两档：
 *
 *   inline（默认，新写法）  标题 + 说明**同行紧跟**（gap-x-3 = 12px），底对齐。
 *                          页头不设下边距（mb-0），间距交给页面容器 `gap-2`（8px）。
 *                          用于**详情页 / 表单页**（它们必须保留页头 = 实体身份）。
 *
 *   stacked（旧写法，兼容） 标题与说明上下堆叠、左右两端对齐、`mb-6`。
 *                          只为未迁移的存量页面保留；新页面不要用。
 *
 * ⚠️ 实测调出来的三个数，别改：
 *   1. `items-end` 不能省 —— 少了它，20px 标题与 14px 说明会变成"中间对齐"，说明会"飘"；
 *   2. 说明必须**紧跟标题右侧 12px** —— 曾用 `justify-between` 推到最右，被否：
 *      「放右边就没效果了」；
 *   3. 说明字号锁 `text-sm`（14px），不要为了配合标题缩小。
 *
 * ⚠️ 列表页**不要用本组件**：列表页去掉页头，内容装 `ContentFrame`。
 *    页头里的按钮（视图切换 / 新建 / 统计计数）要搬到内容区顶部工具行的**最右侧**，
 *    否则"去页头"会把功能一起删掉（这是最容易翻车的一步）。
 */
export interface PageHeaderProps {
  title: string;
  description?: string;
  actions?: React.ReactNode;
  /** 排版：inline = 标题与说明同行（详情/表单页默认）｜stacked = 上下堆叠（存量兼容） */
  layout?: "inline" | "stacked";
  className?: string;
}

export function PageHeader({
  title,
  description,
  actions,
  layout = "inline",
  className,
}: PageHeaderProps) {
  if (layout === "stacked") {
    return (
      <div className={cn("mb-6 flex flex-wrap items-start justify-between gap-4", className)}>
        <div className="flex flex-col gap-1">
          <h1 className="text-xl font-semibold tracking-tight">{title}</h1>
          {description && <p className="text-sm text-muted-foreground">{description}</p>}
        </div>
        {actions && <div className="flex items-center gap-2">{actions}</div>}
      </div>
    );
  }

  return (
    <div
      data-testid="page-header"
      className={cn("mb-0 flex flex-wrap items-end gap-x-3 gap-y-1", className)}
    >
      <h1 className="text-xl font-semibold tracking-tight">{title}</h1>
      {description && <p className="text-sm text-muted-foreground">{description}</p>}
      {actions && <div className="ml-auto flex items-center gap-2">{actions}</div>}
    </div>
  );
}

const pad = (n: number) => String(n).padStart(2, "0");
const ymd = (d?: Date) => (d ? `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}` : "");
const ym = (d?: Date) => (d ? `${d.getFullYear()}-${pad(d.getMonth() + 1)}` : "");
const yOnly = (d?: Date) => (d ? `${d.getFullYear()}` : "");

export interface Shortcut {
  text: string;
  value: Date | (() => Date);
}

const resolveShortcut = (s: Shortcut): Date => (typeof s.value === "function" ? s.value() : s.value);

/* ────────────────────────────────────────────────────────────
 * 通用触发器（按钮外形，左侧图标，整框可点）
 * ★ 必须 forwardRef：PopoverTrigger asChild 需要把 ref 挂到真实 DOM 上，
 *   否则 radix PopperAnchor 拿不到 reference → Floating UI 不定位 → 弹层卡在 -200% 占位。
 * ──────────────────────────────────────────────────────────── */
const Trigger = React.forwardRef<
  HTMLButtonElement,
  {
    display?: string;
    placeholder: string;
    onClear?: () => void;
    className?: string;
    testid: string;
  } & Omit<React.ComponentPropsWithoutRef<typeof Button>, "value" | "onChange" | "className" | "children">
>(function Trigger({ display, placeholder, onClear, className, testid, ...rest }, ref) {
  return (
    <div className={cn("relative w-[240px]", className)}>
      <Button
        ref={ref}
        variant="outline"
        data-testid={testid}
        className={cn("w-full justify-start pr-8 text-left font-normal", !display && "text-muted-foreground")}
        {...rest}
      >
        <CalendarDays className="mr-2 h-4 w-4 shrink-0" />
        <span className="truncate">{display || placeholder}</span>
      </Button>
      {display && onClear && (
        <button
          type="button"
          aria-label="清除"
          onClick={(e) => { e.stopPropagation(); onClear(); }}
          className="absolute right-2 top-1/2 -translate-y-1/2 rounded-sm text-muted-foreground opacity-60 transition-opacity hover:opacity-100"
        >
          <X className="h-3.5 w-3.5" />
        </button>
      )}
    </div>
  );
});

/* ────────────────────────────────────────────────────────────
 * 范围触发器：单个输入框（显示 开始 ~ 结束）+ 右侧清除
 * ★ 必须 forwardRef（PopoverTrigger asChild 需要真实 DOM，否则 radix PopperAnchor 拿不到 reference）
 * ──────────────────────────────────────────────────────────── */
const RangeTrigger = React.forwardRef<
  HTMLDivElement,
  {
    start?: string;
    end?: string;
    startPlaceholder?: string;
    endPlaceholder?: string;
    placeholder?: string;
    onClear?: () => void;
    className?: string;
    testid: string;
  } & Omit<React.ComponentPropsWithoutRef<"div">, "className" | "children">
>(function RangeTrigger(
  { start, end, startPlaceholder = "开始", endPlaceholder = "结束", onClear, className, testid, ...rest },
  ref
) {
  const empty = !start && !end;
  return (
    <div
      ref={ref}
      role="button"
      tabIndex={0}
      data-testid={testid}
      className={cn(
        "relative flex h-9 w-full cursor-pointer items-center gap-2 rounded-md border border-input bg-background px-3 text-sm shadow-sm",
        "ring-offset-background focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2",
        className
      )}
      {...rest}
    >
      <CalendarDays className="h-4 w-4 shrink-0 text-muted-foreground" />
      <span className="flex flex-1 items-center gap-1.5 truncate">
        <span className={cn("truncate", !start && "text-muted-foreground")}>{start || startPlaceholder}</span>
        <span className="shrink-0 text-muted-foreground">~</span>
        <span className={cn("truncate", !end && "text-muted-foreground")}>{end || endPlaceholder}</span>
      </span>
      {!empty && onClear && (
        <button
          type="button"
          aria-label="清除"
          onClick={(e) => { e.stopPropagation(); onClear(); }}
          className="shrink-0 rounded-sm text-muted-foreground opacity-60 transition-opacity hover:opacity-100"
        >
          <X className="h-3.5 w-3.5" />
        </button>
      )}
    </div>
  );
});

/* ────────────────────────────────────────────────────────────
 * 双面板区间选择（左=开始月，右=次月）+ 共享区间状态
 * ★ 左面板选起点，右面板选终点（或相反顺序）；两面板共享“选取中”状态。
 * ──────────────────────────────────────────────────────────── */
function RangePanel({
  selected, onChange, disabledDate, view = "date", onClose, children,
}: {
  selected?: DateRange;
  onChange: (r: DateRange) => void;
  disabledDate?: (d: Date) => boolean;
  view?: CalendarView;
  onClose: () => void;
  children?: React.ReactNode;
}) {
  const anchor = selected?.[0] ?? selected?.[1] ?? new Date();
  const [baseMonth, setBaseMonth] = React.useState(
    () => new Date(anchor.getFullYear(), anchor.getMonth(), 1)
  );
  const [pendingStart, setPendingStart] = React.useState<Date | null>(null);

  const shift = (months: number) => {
    const d = new Date(baseMonth);
    if (view === "date") d.setMonth(d.getMonth() + months);
    else if (view === "month") d.setFullYear(d.getFullYear() + months);
    else d.setFullYear(d.getFullYear() + months * 12);
    setBaseMonth(new Date(d.getFullYear(), d.getMonth(), 1));
  };
  // 年/大跨步（日视图：±1 年；月视图：±10 年；年视图：±10 纪）
  const shiftBig = (dir: number) => {
    const d = new Date(baseMonth);
    if (view === "date") d.setFullYear(d.getFullYear() + dir);
    else if (view === "month") d.setFullYear(d.getFullYear() + dir * 10);
    else d.setFullYear(d.getFullYear() + dir * 120);
    setBaseMonth(new Date(d.getFullYear(), d.getMonth(), 1));
  };

  const pick = (d: Date) => {
    if (!pendingStart) {
      setPendingStart(d);
      onChange([d, undefined]);
      return;
    }
    const [a, b] = pendingStart <= d ? [pendingStart, d] : [d, pendingStart];
    setPendingStart(null);
    onChange([a, b]);
    onClose();
  };

  // 右面板展示的左月 +1
  const rightMonth = React.useMemo(() => {
    const d = new Date(baseMonth);
    if (view === "date") d.setMonth(d.getMonth() + 1);
    else if (view === "month") d.setFullYear(d.getFullYear() + 1);
    else d.setFullYear(d.getFullYear() + 12);
    return d;
  }, [baseMonth, view]);

  return (
    <div className="flex flex-col gap-2" data-testid="range-panel">
      {/* 顶部导航：左 ‹«  ... »› 右（同时推进两个面板） */}
      <div className="flex items-center justify-between px-1">
        <div className="flex items-center gap-0.5">
          <NavArrow onClick={() => shiftBig(-1)} label="上一年" testid="range-prev-big">
            <ChevronsLeft className="h-4 w-4" />
          </NavArrow>
          <NavArrow onClick={() => shift(-1)} label="上一步" testid="range-prev">
            <ChevronLeft className="h-4 w-4" />
          </NavArrow>
        </div>
        <div className="flex items-center gap-0.5">
          <NavArrow onClick={() => shift(1)} label="下一步" testid="range-next">
            <ChevronRight className="h-4 w-4" />
          </NavArrow>
          <NavArrow onClick={() => shiftBig(1)} label="下一年" testid="range-next-big">
            <ChevronsRight className="h-4 w-4" />
          </NavArrow>
        </div>
      </div>
      <div className="flex items-start gap-3">
        <Calendar
          view={view}
          selected={selected}
          onSelectRange={() => {}}
          onPickDate={pick}
          pendingStart={pendingStart}
          edgeRole="start"
          disabledDate={disabledDate}
          baseMonth={baseMonth}
          onMonthChange={setBaseMonth}
          hideArrows
        />
        <Calendar
          view={view}
          selected={selected}
          onSelectRange={() => {}}
          onPickDate={pick}
          pendingStart={pendingStart}
          edgeRole="end"
          disabledDate={disabledDate}
          baseMonth={rightMonth}
          onMonthChange={() => {}}
          hideArrows
        />
      </div>
      {children}
    </div>
  );
}

const NavArrow = ({
  onClick, label, testid, children,
}: { onClick: () => void; label: string; testid: string; children: React.ReactNode }) => (
  <button
    type="button"
    onClick={onClick}
    aria-label={label}
    data-testid={testid}
    className="flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
  >
    {children}
  </button>
);

  value, onChange, placeholder = "选择日期", className, shortcuts, disabledDate, ...rest

  onChange?: (d?: Date) => void;
  placeholder?: string;
  className?: string;
  shortcuts?: Shortcut[];
  disabledDate?: (d: Date) => boolean;

              {shortcuts.map((s) => (
                <Button key={s.text} variant="ghost" size="sm" className="justify-start" onClick={() => { onChange?.(resolveShortcut(s)); setOpen(false); }}>
                  {s.text}
                </Button>
              ))}
            </div>
          ) : null}
          <div>
            <Calendar value={value} disabledDate={disabledDate} onChange={(d) => { onChange?.(d); setOpen(false); }} />
            <div className="mt-2 flex justify-between border-t pt-2">
              <Button variant="ghost" size="sm" onClick={() => { onChange?.(undefined); setOpen(false); }}>清除</Button>
              <Button variant="ghost" size="sm" onClick={() => { onChange?.(new Date()); setOpen(false); }}>今天</Button>
            </div>
          </div>
        </div>
      </PopoverContent>
    </Popover>
  );
}

/* ────────────────────────────────────────────────────────────
 * 7. 日期时间选择器（含时分，支持快捷选项）
 * ──────────────────────────────────────────────────────────── */
export function DateTimePicker({
  value, onChange, className, shortcuts, placeholder = "选择日期时间", ...rest

  value?: Date;
  onChange?: (d?: Date) => void;
  className?: string;
  shortcuts?: Shortcut[];

              {shortcuts.map((s) => (
                <Button key={s.text} variant="ghost" size="sm" className="justify-start" onClick={() => { onChange?.(resolveShortcut(s)); setOpen(false); }}>
                  {s.text}
                </Button>
              ))}
            </div>
          ) : null}
          <div>

/**
 * 轻量日历 —— 零外部依赖，科技感渐变选中态。
 * 相比 react-day-picker：可控性强、体积更小，样式完全走 Tailwind 主题变量。
 *
 * 支持三种视图（view）：
 *   - date  日视图（默认）
 *   - month 月视图（选月份）
 *   - year  年视图（选年份，分页 12 年）
 * 支持区间选择（range）：传入 selected 数组 [start, end]，选中高亮。
 * 支持禁用日期（disabledDate）：返回 true 表示不可选。

export type CalendarView = "date" | "month" | "year";
export type DateRange = [Date | undefined, Date | undefined];

export interface CalendarProps {
  /** 单选值 */
  value?: Date;
  /** 单选回调 */
  onChange?: (d: Date) => void;
  /** 区间值 [start, end]；传入则进入区间选择模式 */
  selected?: DateRange;
  /** 区间回调 */
  onSelectRange?: (r: DateRange) => void;
  /** 视图：日/月/年 */
  view?: CalendarView;
  /** 禁用日期 */
  disabledDate?: (d: Date) => boolean;
  /** 年份跳转步长，默认 1 */
  yearStep?: number;
  /** 受控“当前展示月/年”的基准日期（双面板共享月份用）；不传则内部自管理 */

  /** 展示月份变化回调（配合 baseMonth 做双面板联动） */

  /** 强制控制选中态的“区间端点”角色；仅用于双面板：面板左侧=开始端点，右侧=结束端点 */

  /** 受控“区间选取中”的起始点（双面板共享）；不传则内部自管理 */
  pendingStart?: Date | null;

  /** 额外 className 传给 grid 日格子 */

export function Calendar({
  value = new Date(),
  onChange,
  selected,
  onSelectRange,
  view = "date",
  disabledDate,
  className,
  yearStep = 1,
  baseMonth,
  onMonthChange,
  edgeRole,
  pendingStart,
  onPickDate,
  hideNav,
  hideArrows,
}: CalendarProps) {
  const rangeMode = !!selected;
  const anchor = value ?? selected?.[0] ?? new Date();
  const [innerViewDate, setInnerViewDate] = React.useState(() => new Date(anchor.getFullYear(), anchor.getMonth(), 1));
  // 受控模式：外部传入 baseMonth 则以其为准（双面板共享/联动）
  const viewDate = baseMonth ?? innerViewDate;
  const setViewDate = (d: Date) => {
    if (baseMonth) onMonthChange?.(d);
    else setInnerViewDate(d);
  };
  // 年份视图的分页起点（12 年一屏）
  const [yearBase, setYearBase] = React.useState(() => Math.floor(anchor.getFullYear() / 12) * 12);

  const y = viewDate.getFullYear();
  const m = viewDate.getMonth();
  const today = new Date();

  // 区间选择内部状态（点击第一个点、点击第二个点闭合）
  const [rangePicking, setRangePicking] = React.useState<Date | null>(null);
  React.useEffect(() => {
    if (rangeMode && selected?.[1]) setRangePicking(null);
  }, [rangeMode, selected]);

  /* ── 头部导航 ── */
  const header = (() => {
    if (view === "date") {
      return {
        title: `${y} 年 ${m + 1} 月`,
        titleTestid: "cal-title",
        onPrevMonth: () => setViewDate(new Date(y, m - 1, 1)),
        onNextMonth: () => setViewDate(new Date(y, m + 1, 1)),
        onPrevYear: () => setViewDate(new Date(y - yearStep, m, 1)),
        onNextYear: () => setViewDate(new Date(y + yearStep, m, 1)),
      };
    }
    if (view === "month") {
      return {
        title: `${y} 年`,
        titleTestid: "cal-title",
        onPrevMonth: () => setViewDate(new Date(y - 1, m, 1)),
        onNextMonth: () => setViewDate(new Date(y + 1, m, 1)),
        onPrevYear: () => setViewDate(new Date(y - 10, m, 1)),
        onNextYear: () => setViewDate(new Date(y + 10, m, 1)),
        isYearGroup: true,
      };
    }
    // year
    return {
      title: `${yearBase} - ${yearBase + 11}`,
      titleTestid: "cal-title",
      onPrevMonth: () => setYearBase((b) => b - 12),
      onNextMonth: () => setYearBase((b) => b + 12),
      onPrevYear: () => setYearBase((b) => b - 120),
      onNextYear: () => setYearBase((b) => b + 120),
      isYearGroup: true,
    };
  })();

  /* ── 日视图 cell ── */
  const dayCells: { date: Date; current: boolean }[] = [];
  if (view === "date") {
    const firstDay = new Date(y, m, 1).getDay();
    const daysInMonth = new Date(y, m + 1, 0).getDate();
    const daysInPrev = new Date(y, m, 0).getDate();
    for (let i = firstDay - 1; i >= 0; i--) dayCells.push({ date: new Date(y, m - 1, daysInPrev - i), current: false });
    for (let d = 1; d <= daysInMonth; d++) dayCells.push({ date: new Date(y, m, d), current: true });
    while (dayCells.length % 7 !== 0)
      dayCells.push({ date: new Date(y, m + 1, dayCells.length - firstDay - daysInMonth + 1), current: false });
  }

  const pick = (d: Date) => {
    if (disabledDate?.(d)) return;
    if (onPickDate) {
      // 双面板模式：选取逻辑由父级接管
      if (rangeMode) onSelectRange?.([d, undefined]);
      onPickDate(d);
      return;
    }
    if (rangeMode) {
      if (!rangePicking) {
        setRangePicking(d);
        onSelectRange?.([d, undefined]);
      } else {
        const [a, b] = rangePicking <= d ? [rangePicking, d] : [d, rangePicking];
        setRangePicking(null);
        onSelectRange?.([a, b]);
      }
      return;
    }
    // 月视图：选中整月第一天；年视图：选中该年 1 月 1 日
    onChange?.(d);
  };

  // 区间内高亮判断
  const activePending = pendingStart !== undefined ? pendingStart : rangePicking;
  const start = activePending ?? selected?.[0];
  const end = activePending ? undefined : selected?.[1];
  const inRange = (d: Date) => {
    if (!rangeMode || !start || !end) return false;
    return d > start && d < end;
  };
  const isEdge = (d: Date) => sameDay(d, start) || sameDay(d, end);

  return (
    <div className={cn("w-[268px] select-none", className)}>
      <div className={cn("mb-2 flex items-center justify-between", hideNav && "hidden")}>
        {!hideArrows && (
          <div className="flex items-center gap-0.5">
            <NavBtn onClick={header.onPrevYear} label={header.isYearGroup ? "上十年" : "上一年"} testid="cal-prev-year">
              <ChevronsLeft className="h-4 w-4" />
            </NavBtn>
            <NavBtn onClick={header.onPrevMonth} label={header.isYearGroup ? "上一年" : "上个月"} testid="cal-prev-month">
              <ChevronLeft className="h-4 w-4" />
            </NavBtn>
          </div>
        )}
        <div className={cn("text-sm font-semibold", hideArrows && "mx-auto")} data-testid={header.titleTestid}>{header.title}</div>
        {!hideArrows && (
          <div className="flex items-center gap-0.5">
            <NavBtn onClick={header.onNextMonth} label={header.isYearGroup ? "下一年" : "下个月"} testid="cal-next-month">
              <ChevronRight className="h-4 w-4" />
            </NavBtn>
            <NavBtn onClick={header.onNextYear} label={header.isYearGroup ? "下十年" : "下一年"} testid="cal-next-year">
              <ChevronsRight className="h-4 w-4" />
            </NavBtn>
          </div>
        )}
      </div>

      {/* ── 月视图 ── */}
      {view === "month" && (
        <div className="grid grid-cols-3 gap-1" data-testid="cal-month-grid">
          {MONTHS.map((label, i) => {
            const md = new Date(y, i, 1);
            const active = sameMonth(md, selected?.[0] ?? value);
            const disabled = disabledDate?.(md);
            return (
              <button
                type="button"
                key={label}
                disabled={disabled}
                data-testid={`cal-month-${i + 1}`}
                onClick={() => pick(md)}
                className={cn(
                  "flex h-9 items-center justify-center rounded-md text-sm transition-colors",
                  disabled && "cursor-not-allowed text-muted-foreground/30",
                  active && !disabled && "bg-gradient-to-br from-sky-500 to-indigo-500 font-semibold text-white shadow-[0_0_12px_-2px_rgba(99,102,241,0.8)]",
                  !active && !disabled && "hover:bg-accent"
                )}
              >
                {label}
              </button>
            );
          })}
        </div>
      )}

      {/* ── 年视图 ── */}
      {view === "year" && (
        <div className="grid grid-cols-3 gap-1" data-testid="cal-year-grid">
          {Array.from({ length: 12 }, (_, i) => yearBase + i).map((yy) => {
            const yd = new Date(yy, 0, 1);
            const active = sameYear(yd, selected?.[0] ?? value);
            const disabled = disabledDate?.(yd);
            return (
              <button
                type="button"
                key={yy}
                disabled={disabled}
                data-testid={`cal-year-${yy}`}
                onClick={() => pick(yd)}
                className={cn(
                  "flex h-9 items-center justify-center rounded-md text-sm transition-colors",
                  disabled && "cursor-not-allowed text-muted-foreground/30",
                  active && !disabled && "bg-gradient-to-br from-sky-500 to-indigo-500 font-semibold text-white shadow-[0_0_12px_-2px_rgba(99,102,241,0.8)]",
                  !active && !disabled && "hover:bg-accent"
                )}
              >
                {yy}
              </button>
            );
          })}
        </div>
      )}

      {/* ── 日视图 ── */}
      {view === "date" && (
        <>
          <div className="grid grid-cols-7 gap-0.5 text-center text-[11px] text-muted-foreground">
            {WEEK.map((w) => <div key={w} className="py-1">{w}</div>)}
          </div>
          <div className="grid grid-cols-7 gap-0.5">
            {dayCells.map((c, i) => {
              const disabled = disabledDate?.(c.date);
              const active = rangeMode
                ? (edgeRole === "start" && !activePending
                    ? sameDay(c.date, start)
                    : edgeRole === "start" && activePending
                    ? sameDay(c.date, activePending)
                    : edgeRole === "end"
                    ? sameDay(c.date, end)
                    : isEdge(c.date))
                : sameDay(c.date, value);
              const ranged = inRange(c.date);
              const isToday = sameDay(c.date, today);
              return (
                <button
                  type="button"
                  key={i}
                  disabled={disabled}
                  onClick={() => pick(c.date)}
                  className={cn(
                    "flex h-8 items-center justify-center rounded-md text-xs transition-colors",
                    !c.current && "text-muted-foreground/40",
                    disabled && "cursor-not-allowed text-muted-foreground/25 line-through",
                    active && !disabled && "bg-gradient-to-br from-sky-500 to-indigo-500 font-semibold text-white shadow-[0_0_12px_-2px_rgba(99,102,241,0.8)]",
                    ranged && !disabled && "bg-accent text-foreground",
                    !active && !ranged && !disabled && "hover:bg-accent",
                    isToday && !active && !disabled && "ring-1 ring-primary/50"
                  )}
                >
                  {c.date.getDate()}
                </button>
              );
            })}
          </div>
        </>
      )}
    </div>
  );
}

import * as React from "react";
import { ChevronRight, ChevronLeft, Search } from "lucide-react";
import { cn } from "@/lib/utils";

export interface TransferItem {
  key: string;
  title: string;
  description?: string;
  disabled?: boolean;
}

/**
 * 穿梭框（Transfer）—— shadcn 风格自研实现。
 * 左：可选项；右：已选项。支持搜索、全选、禁用项、统计。
 */
export function Transfer({
  dataSource,
  targetKeys: targetKeysProp,
  onChange,
  titles = ["可选项", "已选项"],
  className,
  height = 260,
}: {
  dataSource: TransferItem[];
  targetKeys?: string[];
  onChange?: (keys: string[]) => void;
  titles?: [string, string];
  className?: string;
  height?: number;
}) {
  const [targetKeys, setTargetKeys] = React.useState<string[]>(targetKeysProp ?? []);
  const [leftChecked, setLeftChecked] = React.useState<string[]>([]);
  const [rightChecked, setRightChecked] = React.useState<string[]>([]);
  const [leftQuery, setLeftQuery] = React.useState("");
  const [rightQuery, setRightQuery] = React.useState("");

  React.useEffect(() => {
    if (targetKeysProp) setTargetKeys(targetKeysProp);
  }, [targetKeysProp]);

  const commit = (keys: string[]) => {
    setTargetKeys(keys);
    onChange?.(keys);
  };

  const source = dataSource.filter((i) => !targetKeys.includes(i.key));
  const target = dataSource.filter((i) => targetKeys.includes(i.key));

  const filter = (list: TransferItem[], q: string) =>
    list.filter(
      (i) =>
        i.title.toLowerCase().includes(q.toLowerCase()) ||
        (i.description ?? "").toLowerCase().includes(q.toLowerCase())
    );

  const leftList = filter(source, leftQuery);
  const rightList = filter(target, rightQuery);

  const moveRight = () => {
    const moved = leftChecked.filter((k) => !dataSource.find((d) => d.key === k)?.disabled);
    commit([...targetKeys, ...moved]);
    setLeftChecked([]);
  };
  const moveLeft = () => {
    commit(targetKeys.filter((k) => !rightChecked.includes(k)));
    setRightChecked([]);
  };

  return (
    <div className={cn("flex flex-col gap-3 sm:flex-row sm:items-stretch", className)}>
      <Panel
        title={titles[0]}
        count={leftList.length}
        total={source.length}
        list={leftList}
        checked={leftChecked}
        onCheck={setLeftChecked}
        query={leftQuery}
        onQuery={setLeftQuery}
        height={height}
      />
      <div className="flex flex-row items-center justify-center gap-2 sm:flex-col">
        <Button size="icon" variant="outline" onClick={moveRight} disabled={leftChecked.length === 0} title="移至右侧">
          <ChevronRight className="h-4 w-4" />
        </Button>
        <Button size="icon" variant="outline" onClick={moveLeft} disabled={rightChecked.length === 0} title="移至左侧">
          <ChevronLeft className="h-4 w-4" />
        </Button>
      </div>
      <Panel
        title={titles[1]}
        count={rightList.length}
        total={target.length}
        list={rightList}
        checked={rightChecked}
        onCheck={setRightChecked}
        query={rightQuery}
        onQuery={setRightQuery}
        height={height}
      />
    </div>
  );
}

function Panel({
  title, count, total, list, checked, onCheck, query, onQuery, height,
}: {
  title: string;
  count: number;
  total: number;
  list: TransferItem[];
  checked: string[];
  onCheck: (k: string[]) => void;
  query: string;
  onQuery: (q: string) => void;
  height: number;
}) {
  const selectable = list.filter((i) => !i.disabled);
  const allChecked = selectable.length > 0 && selectable.every((i) => checked.includes(i.key));
  const toggleAll = () => {
    if (allChecked) onCheck(checked.filter((k) => !selectable.some((i) => i.key === k)));
    else onCheck(Array.from(new Set([...checked, ...selectable.map((i) => i.key)])));
  };
  const toggleOne = (key: string) =>
    onCheck(checked.includes(key) ? checked.filter((k) => k !== key) : [...checked, key]);

  return (
    <div className="flex flex-1 flex-col rounded-lg border bg-card/50">
      <div className="flex items-center justify-between border-b px-3 py-2">
        <label className="flex items-center gap-2 text-xs font-medium">
          <Checkbox checked={allChecked} onCheckedChange={toggleAll} />
          {title}
          <span className="text-muted-foreground">{checked.length}/{total}</span>
        </label>
        <span className="text-[11px] text-muted-foreground">{count} 项</span>
      </div>
      <div className="border-b px-3 py-1.5">
        <div className="relative">
          <Search className="absolute left-2 top-1.5 h-3.5 w-3.5 text-muted-foreground" />
          <input
            value={query}
            onChange={(e) => onQuery(e.target.value)}
            placeholder="搜索…"
            className="h-7 w-full rounded-md border border-input bg-background/60 pl-7 pr-2 text-xs placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
          />
        </div>
      </div>
      <div className="overflow-y-auto p-1" style={{ height }}>
        {list.length === 0 && <div className="py-8 text-center text-xs text-muted-foreground">无数据</div>}
        {list.map((i) => (
          <label
            key={i.key}
            className={cn(
              "flex cursor-pointer items-center gap-2 rounded-md px-2 py-1.5 text-sm transition-colors",
              i.disabled ? "cursor-not-allowed opacity-40" : "hover:bg-accent"
            )}
          >
            <Checkbox
              disabled={i.disabled}
              checked={checked.includes(i.key)}
              onCheckedChange={() => toggleOne(i.key)}
            />
            <span className="flex min-w-0 flex-col">
              <span className="truncate">{i.title}</span>
              {i.description ? (
                <span className="truncate text-[11px] text-muted-foreground">{i.description}</span>
              ) : null}
            </span>
          </label>
        ))}
      </div>
    </div>
  );
}

/**
 * 富文本编辑器（Tiptap / ProseMirror 内核）。
 *
 * 选型理由：
 * - Tiptap 基于 ProseMirror，headless、对 React 友好，样式完全由 Tailwind 控制 → 与 shadcn 体系无缝

import { useEditor, EditorContent, type Editor } from "@tiptap/react";
import StarterKit from "@tiptap/starter-kit";
import Link from "@tiptap/extension-link";
import Image from "@tiptap/extension-image";

const Sep = () => <span className="mx-1 h-4 w-px bg-border" />;

function Toolbar({ editor, onPickImage }: { editor: Editor; onPickImage: () => void }) {
  if (!editor) return null;
  const addLink = () => {
    const url = window.prompt("链接地址");
    if (url) editor.chain().focus().extendMarkRange("link").setLink({ href: url }).run();
    else editor.chain().focus().unsetLink().run();
  };
  return (
    <div className="flex flex-wrap items-center gap-0.5 border-b bg-muted/30 px-2 py-1.5">
      <ToolbarButton title="撤销" onClick={() => editor.chain().focus().undo().run()}><Undo2 className="h-3.5 w-3.5" /></ToolbarButton>
      <ToolbarButton title="重做" onClick={() => editor.chain().focus().redo().run()}><Redo2 className="h-3.5 w-3.5" /></ToolbarButton>
      <Sep />
      <ToolbarButton title="H1" active={editor.isActive("heading", { level: 1 })} onClick={() => editor.chain().focus().toggleHeading({ level: 1 }).run()}><Heading1 className="h-3.5 w-3.5" /></ToolbarButton>
      <ToolbarButton title="H2" active={editor.isActive("heading", { level: 2 })} onClick={() => editor.chain().focus().toggleHeading({ level: 2 }).run()}><Heading2 className="h-3.5 w-3.5" /></ToolbarButton>
      <ToolbarButton title="H3" active={editor.isActive("heading", { level: 3 })} onClick={() => editor.chain().focus().toggleHeading({ level: 3 }).run()}><Heading3 className="h-3.5 w-3.5" /></ToolbarButton>
      <Sep />
      <ToolbarButton title="加粗" active={editor.isActive("bold")} onClick={() => editor.chain().focus().toggleBold().run()}><Bold className="h-3.5 w-3.5" /></ToolbarButton>
      <ToolbarButton title="斜体" active={editor.isActive("italic")} onClick={() => editor.chain().focus().toggleItalic().run()}><Italic className="h-3.5 w-3.5" /></ToolbarButton>
      <ToolbarButton title="下划线" active={editor.isActive("underline")} onClick={() => editor.chain().focus().toggleUnderline().run()}><UnderlineIcon className="h-3.5 w-3.5" /></ToolbarButton>
      <ToolbarButton title="删除线" active={editor.isActive("strike")} onClick={() => editor.chain().focus().toggleStrike().run()}><Strikethrough className="h-3.5 w-3.5" /></ToolbarButton>
      <ToolbarButton title="行内代码" active={editor.isActive("code")} onClick={() => editor.chain().focus().toggleCode().run()}><Code className="h-3.5 w-3.5" /></ToolbarButton>
      <Sep />

      <ToolbarButton title="有序列表" active={editor.isActive("orderedList")} onClick={() => editor.chain().focus().toggleOrderedList().run()}><ListOrdered className="h-3.5 w-3.5" /></ToolbarButton>

export function RichTextEditor({
  value = "",
  onChange,
  placeholder = "在此输入内容…",
  className,
  minHeight = 220,
}: {
  value?: string;
  onChange?: (html: string) => void;
  placeholder?: string;
  className?: string;
  minHeight?: number;
}) {
  const editor = useEditor({
    extensions: [
      StarterKit.configure({ heading: { levels: [1, 2, 3] } }),
      Underline,
      Link.configure({ openOnClick: false, HTMLAttributes: { class: "text-primary underline" } }),
      Image.configure({ inline: false, allowBase64: true, HTMLAttributes: { class: "rounded-md" } }),
      Placeholder.configure({ placeholder }),
      TextAlign.configure({ types: ["heading", "paragraph"] }),
    ],
    content: value,
    onUpdate: ({ editor }) => onChange?.(editor.getHTML()),
    editorProps: {
      attributes: {
        class: "prose-win3 focus:outline-none",
        style: `min-height:${minHeight}px`,
      },
    },
  });

  const fileRef = React.useRef<HTMLInputElement>(null);

  // 图片上传：读取为 base64 并插入（对齐 vue-admin-tpl 的 uploadImgShowBase64，纯前端 demo 不落库）
  const handleFile = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file || !editor) return;
    const reader = new FileReader();
    reader.onload = () => {
      editor.chain().focus().setImage({ src: String(reader.result), alt: file.name }).run();
    };
    reader.readAsDataURL(file);
    e.target.value = ""; // 允许重复选同一文件
  };

  // 外部 value 变化时同步（避免受控组件卡死）
  React.useEffect(() => {
    if (editor && value !== editor.getHTML()) {
      editor.commands.setContent(value || "", { emitUpdate: false });
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [value]);

  return (
    <div className={cn("overflow-hidden rounded-lg border bg-background/60", className)}>
      <input
        ref={fileRef}
        type="file"
        accept="image/*"
        className="hidden"
        data-testid="rte-image-input"
        onChange={handleFile}
      />
      {editor && <Toolbar editor={editor} onPickImage={() => fileRef.current?.click()} />}
      <div className="px-3 py-2">
        <EditorContent editor={editor} />
      </div>
    </div>
  );
}

/**
 * Markdown 编辑器（编辑 + 实时预览）。
 *
 * 选型理由：
 * - @uiw/react-md-editor：开箱即用，自带分栏预览 / 工具栏 / 语法高亮，
 *   基于 rehype/remark 生态，社区成熟，体积可控。
 */
import * as React from "react";
import MDEditor from "@uiw/react-md-editor";
import { cn } from "@/lib/utils";

export function MarkdownEditor({
  value,
  onChange,
  height = 360,
  className,
}: {
  value?: string;
  onChange?: (v: string) => void;
  height?: number;
  className?: string;
}) {
  const [dark, setDark] = React.useState(() =>
    typeof document !== "undefined" ? document.documentElement.classList.contains("dark") : true
  );

  // 跟随站点主题
  React.useEffect(() => {
    const el = document.documentElement;
    const observer = new MutationObserver(() => setDark(el.classList.contains("dark")));
    observer.observe(el, { attributes: true, attributeFilter: ["class"] });
    return () => observer.disconnect();
  }, []);

  return (
    <div
      className={cn("win3-md overflow-hidden rounded-lg border bg-background/60", className)}
      data-color-mode={dark ? "dark" : "light"}
    >
      <MDEditor
        value={value}
        onChange={(v) => onChange?.(v ?? "")}
        height={height}
        preview="live"
        visibleDragbar={false}
        style={{ background: "transparent" }}
      />
    </div>
  );
}
